8. Structs

结构体

结构体用于把多项相关数据组织成一个自定义类型。本章整理结构体定义、成员访问、结构体数组、指针和链表基础。

结构体

图示转写：C++ 类型分类

C++ 中的数据类型可以粗略分为两类：

类别示例
基本类型整型、实型、字符型、布尔型、枚举等
非基本类型数组、指针、引用、结构体、联合、类等

结构体属于用户自定义类型，适合把一组相关数据组织成一个整体。

初始化

法一

struct studentT {
    const char *id;
    const char *name;
    int score;
};

int main()
{
    studentT student1;                   //student1在栈区
    student1.id = "00315";               //"00315"在常量区，id指向它
    student1.name = "San Zhang";         //"San Zhang"在常量区，name指向它
    student1.score = 99;
    cout << student1.name << endl;
}

┌────────────────────────────────────────────────────────┐
│                     常量区（只读）                       │
│  ┌──────────┐  ┌──────────────┐                        │
│  │ "00315"  │  │ "San Zhang"  │                        │
│  └──────────┘  └──────────────┘                        │
│       ↑              ↑                                 │
└───────┼──────────────┼─────────────────────────────────┘
        │              │
        │              │
┌───────┼──────────────┼─────────────────────────────────┐
│       │     栈区      │                                 │
│  ┌────┴──────┐ ┌─────┴─────┐                           │
│  │ id 指针    │ │ name 指针 │  ← student1 在栈区         │
│  │ 0x1000    │ │ 0x2000    │                           │
│  └───────────┘ └───────────┘                           │
│  ┌───────────┐                                         │
│  │ score=99  │                                         │
│  └───────────┘                                         │
└────────────────────────────────────────────────────────┘
  • id 和 name 是 指向常量字符的指针（const char*）
  • 字符串字面量 "00315" 和 "San Zhang" 存储在常量区，类型是 const char[]，可以隐式转换为 const char*
  • 赋值操作只是让指针指向常量区的字符串，不需要拷贝字符数据，也不需要分配内存

法二

struct studentT {
    char *id;
    char *name;
    int score;
};

int main()
{
    studentT student1;
    student1.id = new char[20];
    student1.name = new char[50];
    strncpy(student1.id, "00315", 19);
    strncpy(student1.name, "San Zhang", 49);
    student1.score = 99;
    cout << student1.name << endl;
    cout << student1.id << endl;
    cout << student1.score << endl;
    delete[] student1.id;
    delete[] student1.name;
}

图示转写：定义结构体类型的变量

结构体类型定义后，可以像基本类型一样定义变量：

struct student {
    char id[14];
    char name[12];
    int math;
    int english;
    int programming;
};

student s1;
student students[10];
student* p;

结构体变量在内存中通常是一块连续空间，但成员之间可能因为内存对齐而存在填充字节。可以使用 #pragma pack(n) 改变对齐方式，但一般不建议随意修改。

结构体的总大小，大于等于所有成员变量的内存总和，等于其成员中所含最大类型的整数倍

结构变量的赋值

同类型的结构变量之间可以相互赋值，如：

students1 = students2;  //将student2的成员变量对应的赋值给student1的相应成员变量

相当于：

strcpy(student1.id, student2.id);
strcpy(student1.name, student2.name); //注意不是直接相等赋值！
student1.math = student2.math;
student1.english = student2.english;
student1.programming = student2.programming;

回顾：变量初始化的方法

变量初始化：略

数组初始化：略

内存的动态分配:

int *p = new int;	        // 动态分配一个整型变量
int *q = new int(10);      // 初始化动态分配的变量
int *array1 = new int[10]; // 动态分配整型数组，随机值
//C++/C不允许对动态数组初始化，除了默认初始值0
int *array2 = new int[10]();//初始值为0

结构体变量使用点运算符访问成员：

student s = {"00315", "San Zhang", 11, 28, 2006};
s.math = 98;

student* p = &s;

注意：const char *id和const char *name，需要有const，才能运行student1.id = "00315"

如果有const，则运行strcpy(student1.id, "00315")会报错，因为是常量

#include <iostream>
#include <cstring>
using namespace std;

struct studentT {
    const char *id;
    const char *name;
    int score;
};

int main()
{
    studentT student1;
    student1.id = "00315";
    student1.name = "San Zhang";
    student1.score = 99;
    cout << student1.name << endl;

    studentT student2 = {"00316", "Si Li", 88};
    cout << student2.name << endl;
    return 0;
}

结构体与函数

bool fetch(const DoubleArray &arr, int index, double &value);
void cleanup(const DoubleArray &arr); //回收数组空间

如果函数不需要修改结构体变量，建议加上const

通常采用引用传递

返回结构体类型的函数

#include <iostream>
#include <cstring>
using namespace std;

struct studentT
{
    char *id;
    char *name;
    int score;
};

// 返回结构体的函数
studentT getStudentData()
{
    studentT student;
    student.id = new char[100];
    student.name = new char[100];
    cin.getline(student.id, 100);
    cin.getline(student.name, 100);
    cin >> student.score;    
    return student; // 返回结构体的副本
}

int main()
{
    studentT s1;
    s1 = getStudentData(); // 接收返回的结构体副本

    cout << "ID: " << s1.id << endl;
    cout << "Name: " << s1.name << endl;
    cout << "Score: " << s1.score << endl;
    delete[] s1.id;
    delete[] s1.name;
    return 0;
}

返回结构体引用的函数

#include <iostream>
#include <cstring>
using namespace std;

struct studentT {
    char id[10];
    const char *name;
    int score;
};

// 返回结构体引用
studentT& getStudentData() {
    studentT* student = new studentT;  // 在堆区动态分配
    strcpy(student->id, "00315");
    student->name = "San Zhang";
    student->score = 99;
    return *student;  // 返回解引用的对象（引用）
}

int main() {
    studentT& s1 = getStudentData();  // s1 是引用，指向堆区对象
    
    cout << "ID: " << s1.id << endl;
    cout << "Name: " << s1.name << endl;
    cout << "Score: " << s1.score << endl;
    
    // 不需要时，必须手动释放！
    delete &s1;  // 取地址，删除堆区对象
    
    return 0;
}

链表

单链表的操作：

插入一个结点

ListNode *tmp;
tmp = new ListNode;
tmp->data = x;
tmp->next = p->next;
p->next = tmp;

最后两步顺序不能颠倒

删除一个结点

delPtr = p->next;
p->next = delPtr->next;
delete delPtr;

单链表的建立与访问

正序：

#include <iostream>
#include <cstring>
using namespace std;

struct linkRec {
    int data;
    linkRec *next;
};

int main() {
    int x;  //存放输入的值
    linkRec *head, *rear; //head指向链表头结点，rear指向链表尾结点
    head = rear = new linkRec; //创建空链表，head和rear都指向头结点

    //创建链表的其他结点
    while (true)
    {
        cin >> x;
        if (x == 0) break; //输入0时结束输入
        rear->next = new linkRec; //创建新结点，并将其连接到链表尾部
        rear = rear->next; //新结点作为新的链表尾结点
        rear->data = x; //将输入的值存储在新结点中
        rear->next = nullptr; //新结点的next指针指向nullptr，表示链表结束
    }

    //读取链表中的数据并输出
    rear = head->next; //从链表的第一个结点开始读取数据
    while (rear != nullptr)
    {
        cout << rear->data << ' '; //输出当前结点的数据
        rear = rear->next; //移动到下一个结点
    }

    rear = head->next;
    linkRec *temp;
    while (rear != nullptr)
    {
        temp = rear;
        rear = rear->next;
        delete temp; //释放当前结点的内存
    }
    delete head; //释放头结点的内存
    return 0;
}

temp 是一个指针变量，它本身存储在栈区。delete temp 释放的是 temp 指向的内存（堆区），但**temp 这个指针变量本身依然存在**，可以继续被赋值。

逆序：

#include <iostream>

using namespace std;

struct List
{
    int num;
    List *next;
};

int main()
{
    List *head, *p, *q;
    int num;

    head = NULL;
    cin >> num;
    while (num != -1)
    {
        p = new List;
        p->num = num;
        p->next = head;
        head = p;
        cin >> num;
    }

    // 此处完成链表的遍历及输出("按照输入顺序逆序输出\n")
    for (p = head; p != NULL; p = q)
    {
        cout << p->num << " ";
        q = p->next;
    }

    cout << endl;
    for (p = head; p != NULL; p = q)
    {
        q = p->next;
        delete p;
    }

    return 0;
}
